  329|     if not path.exists():

  332|     for line in path.read_text(encoding="utf-8").splitlines():

  465| def knowledge_stats(knowledge_root: Optional[Path] = None) -> dict[str, int]:
  466|     root = Path(knowledge_root) if knowledge_root is not None else DEFAULT_KNOWLEDGE_ROOT
  467|     point_codes: set[str] = set()
  468|     trackable_codes: set[str] = set()
  469|     subjects: set[str] = set()
  470|     published = 0
  471|     index_files = 0
  472|     if not root.exists():
  473|         return {
  474|             "point_nodes": 0,
  475|             "trackable_cards": 0,
  476|             "subjects": 0,
  477|             "published": 0,
  478|             "index_files": 0,
  479|         }
  480|     for path in root.rglob("*.md"):
  481|         if path.name == "INDEX.md":
  482|             index_files += 1
  483|         metadata = _read_frontmatter(path)
  484|         point_code = str(metadata.get("point_code", "")).strip()
  485|         if not point_code:
  486|             continue
  487|         point_codes.add(point_code)
  488|         subject = str(metadata.get("subject", "")).strip()
  489|         if subject:
  490|             subjects.add(subject)
  491|         if metadata.get("mastery_granularity") == "knowledge_point":

  504| def match_question(
  505|     subject: str,
  506|     grade: str,
  507|     text: str,
  508|     *,
  509|     knowledge_root: Optional[Path] = None,
  510| ) -> MatchResult:
  511|     candidates = [
  512|         point
  513|         for point in registry(knowledge_root)
  514|         if point.subject == subject and point.grade == grade
  515|     ]
  516|     if not candidates:
  517|         return MatchResult(
  518|             point=None,
  519|             confidence=0.0,
  520|             reason=f"没有已发布的{subject}{grade}知识点",
  521|             candidates=[],
  522|             review_required=True,
  523|         )
  524|
  525|     normalized_text = _normalize(text)
  526|     ranked: list[MatchCandidate] = []
  527|     for point in candidates:
  528|         terms = [term for term in point.matching_keywords if _normalize(term) in normalized_text]
  529|         title_match = bool(point.title and _normalize(point.title) in normalized_text)
  530|         if not terms and not title_match:
  531|             continue
  532|         confidence = 0.95 if title_match else min(0.92, 0.72 + 0.06 * (len(terms) - 1))
  533|         ranked.append(MatchCandidate(point=point, confidence=confidence, matched_terms=terms))
  534|
  535|     ranked.sort(key=lambda item: (-item.confidence, -max((len(term) for term in item.matched_terms), default=0), item.point.point_code))
  536|     if not ranked:
  537|         return MatchResult(
  538|             point=None,
  539|             confidence=0.0,
  540|             reason="未命中已发布知识点的匹配关键词",
  541|             candidates=[],
  542|             review_required=True,
  543|         )
  544|
  545|     best = ranked[0]
  546|     ambiguous = len(ranked) > 1 and best.confidence - ranked[1].confidence < 0.1
  547|     terms = "、".join(best.matched_terms) if best.matched_terms else best.point.title
  548|     return MatchResult(
  549|         point=best.point,
  550|         confidence=best.confidence,
  551|         reason=f"命中关键词：{terms}",
  552|         candidates=ranked[:5],
  553|         review_required=ambiguous,
  554|     )

  557| def _read_frontmatter(path: Path) -> dict[str, object]:
  558|     lines = path.read_text(encoding="utf-8").splitlines()
  559|     if not lines or lines[0].strip() != "---":
  560|         return {}
  561|     metadata: dict[str, object] = {}
  562|     current_list: Optional[str] = None
  563|     for line in lines[1:]:
  564|         if line.strip() == "---":
  565|             break
  566|         list_item = re.match(r"^\s+-\s+(.*)$", line)
  567|         if list_item and current_list:
  568|             value = _clean_value(list_item.group(1))
  569|             items = metadata.setdefault(current_list, [])
  570|             if isinstance(items, list):
  571|                 items.append(value)
  572|             continue
  573|         match = re.match(r"^([A-Za-z0-9_]+):\s*(.*)$", line)
  574|         if not match:
  575|             current_list = None
  576|             continue
  577|         key, raw_value = match.group(1), match.group(2).strip()
  578|         if not raw_value:
  579|             metadata[key] = []
  580|             current_list = key
  581|         elif raw_value == "[]":
  582|             metadata[key] = []
  583|             current_list = None
  584|         else:
  585|             metadata[key] = _clean_value(raw_value)
  586|             current_list = None
  587|     return metadata

  591|     return value.strip().strip('"').strip("'")

  595|     if isinstance(value, list):
  596|         return [str(item) for item in value]
  597|     if not value:
  598|         return []
  599|     return [str(value)]

 3269| def _parse_multipart(content_type: str, body: bytes) -> tuple[dict[str, str], Optional[tuple[str, bytes\]\]]:
 3270|     raw = f"Content-Type: {content_type}\r\nMIME-Version: 1.0\r\n\r\n".encode() + body
 3271|     message = BytesParser(policy=default).parsebytes(raw)
 3272|     fields: dict[str, str] = {}
 3273|     upload: Optional[tuple[str, bytes\]\] = None
 3274|     for part in message.iter_parts():
 3275|         name = part.get_param("name", header="content-disposition")
 3276|         if not name:
 3277|             continue
 3278|         filename = part.get_filename()
 3279|         payload = part.get_payload(decode=True) or b""
 3280|         if filename:
 3281|             upload = (filename, payload)
 3282|         else:
 3283|             fields[name] = payload.decode(part.get_content_charset() or "utf-8", errors="replace").strip()
 3284|     return fields, upload
